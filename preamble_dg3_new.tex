% theorem style
\newtheoremstyle{thmred}
  {0pt}                        % space above
  {0pt}                        % space below
  {\normalfont}                % body font (roman, not italic)
  {0pt}                        % indent
  {\bfseries\color{boxred}} % header font
  {.}                          % punctuation after header
  {0.5em}                      % space after header
  {}                           % header spec (empty=default)

\newtheoremstyle{thmgrey}
  {0pt}
  {0pt}
  {\normalfont}
  {0pt}
  {\bfseries\color{boxgrey}}
  {.}
  {0.5em}
  {}

\newtheoremstyle{thmblue}
  {0pt}
  {0pt}
  {\normalfont}
  {0pt}
  {\bfseries\color{boxblue}}
  {.}
  {0.5em}
  {}

  {0pt}                        % indent

  {0.5em}                      % space after header
  {}                           % header spec
